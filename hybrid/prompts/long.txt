The harbor town had grown up around a single stone quay, built in the years when the river still ran deep enough for ocean ships to come upstream. Merchants from three kingdoms met there each spring, and the customs house kept careful ledgers of everything that passed: salt from the southern marshes, iron from the hill country, wool, dyed cloth, glass beads, and barrels of a sour red wine that nobody admitted to liking but everybody bought. When the river began to silt in the second century of the town's life, the council argued for eleven years about whether to dredge it. The dredging faction pointed to the ledgers, which showed that nearly half the town's income came from ships too large to reach any other port on that coast. The other faction pointed out that dredging would cost more than the town earned in a decade, and that the silt came back every winter no matter what anyone did. In the end nothing was decided, and the question answered itself: the ships simply stopped coming. Within a generation the quay was a place where children fished for eels, the customs house became a granary, and the ledgers were sold by the pound to a bookbinder who used their vellum pages to stiffen the covers of prayer books. Historians who study the region today rely heavily on those covers, because the fragments bound inside them are the only surviving record of prices, tariffs, and the names of ordinary traders from that period. What the council could not save by argument, the bookbinder preserved by accident, and the town's most lasting contribution to history turned out to be its bureaucracy rather than its trade.
import heapq
from dataclasses import dataclass, field


@dataclass(order=True)
class Task:
    priority: int
    name: str = field(compare=False)
    deadline: float = field(compare=False, default=float("inf"))
    retries: int = field(compare=False, default=0)


class Scheduler:
    """Runs the most urgent task first, retrying failures with a penalty."""

    def __init__(self, max_retries=3, penalty=5):
        self._heap = []
        self.max_retries = max_retries
        self.penalty = penalty
        self.completed = []
        self.dropped = []

    def submit(self, task):
        heapq.heappush(self._heap, task)

    def run(self, handler, now=0.0):
        while self._heap:
            task = heapq.heappop(self._heap)
            if now > task.deadline:
                self.dropped.append((task.name, "missed deadline"))
                continue
            try:
                elapsed = handler(task)
            except Exception as exc:  # the handler decides what counts as failure
                if task.retries + 1 >= self.max_retries:
                    self.dropped.append((task.name, repr(exc)))
                else:
                    task.retries += 1
                    task.priority += self.penalty
                    heapq.heappush(self._heap, task)
                continue
            now += elapsed
            self.completed.append((task.name, now))
        return self.completed, self.dropped


def flaky_handler(task):
    if task.name.startswith("net") and task.retries < 1:
        raise TimeoutError("connection reset")
    return 0.5 if task.priority < 10 else 1.5


if __name__ == "__main__":
    s = Scheduler()
    for i, name in enumerate(["net-fetch", "parse", "index", "net-upload", "report"]):
        s.submit(Task(priority=i * 3, name=name, deadline=6.0))
    done, dropped = s.run(flaky_handler)
    for name, t in done:
        print(f"{name:12s} finished at {t:4.1f}")
    for name, why in dropped:
        print(f"{name:12s} dropped: {why}")
